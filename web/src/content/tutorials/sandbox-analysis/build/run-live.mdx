This path requires Docker and an API key. Keep both values outside agent definitions. Set `PURISTA_DOCKER_SANDBOX_IMAGE` to the inspected immutable digest, never a mutable tag.

```dotenv title=".env.example" write
OPENAI_API_KEY=
OPENAI_MODEL=gpt-5-mini
PURISTA_DOCKER_SANDBOX_IMAGE=
PURISTA_DOCKER_SANDBOX_ROOT=./runtime/sandboxes
```
```ts title="src/index.ts" write
import { resolve } from 'node:path'
import { DefaultEventBridge, gracefulShutdown, initLogger } from '@purista/core'
import { openai } from '@purista/harness-openai'
import { dockerSandbox } from '@purista/harness-sandbox-docker'
import { createAnalysisService } from './createAnalysisService.js'
import { transactionAnalysisSandboxCleanup } from './dockerSandboxCleanup.js'

async function main() {
	const logger = initLogger()
	const apiKey = process.env.OPENAI_API_KEY?.trim()
	const image = process.env.PURISTA_DOCKER_SANDBOX_IMAGE?.trim()
	if (!apiKey) throw new Error('OPENAI_API_KEY is required to start the optional live example.')
	if (!image) throw new Error('PURISTA_DOCKER_SANDBOX_IMAGE is required to start the optional live example.')
	const root = resolve(process.env.PURISTA_DOCKER_SANDBOX_ROOT?.trim() || './runtime/sandboxes')
	const sandbox = dockerSandbox({
		root,
		image,
		user: '10001:10001',
	})
	const eventBridge = new DefaultEventBridge({ logger })
	await eventBridge.start()
	const service = await createAnalysisService(eventBridge, logger, {
		analysisPolicy: {
			canRun: async ({ tenantId, principalId }) => tenantId === 'tenant-example' && principalId === 'principal-analyst',
		},
		analysisModel: {
			provider: openai({ apiKey }),
			model: process.env.OPENAI_MODEL?.trim() || 'gpt-5-mini',
		},
		sandbox,
	})
	await service.start()
	gracefulShutdown(logger, [service, transactionAnalysisSandboxCleanup(sandbox.administration, root), eventBridge])
	logger.info('Sandbox transaction analysis service started')
}

main().catch((error: unknown) => {
	process.stderr.write(`${error instanceof Error ? error.message : 'Sandbox analysis service could not start.'}\n`)
	process.exit(1)
})
```
The shutdown registration contains `[service, sandbox cleanup, eventBridge]`. Purge installs revocation state first; the helper removes this disposable metadata root only after cleanup completes, so a later live demo starts clean. The Docker adapter enforces the container boundary. The agent still receives only its narrow Harness tool allowlist.

```bash title="Start the optional live service" replay="project"
npm run start
```
