Start in an empty parent directory. This chapter creates an independent project.

```bash title="Create and enter the project" replay="parent"
npm create purista@3.0.0 multi-step-support-workflow -- --runtime node --event-bridge default --non-interactive --defaults
cd multi-step-support-workflow
```

```bash title="Install the project dependencies" replay="project"
npm install
```

Generate each boundary once. `--http none` keeps both agents internal. The command target is `runResolveSupportCase`, while the workflow target is `resolveSupportCase`.

```bash title="Generate the Support boundaries" replay="project"
npm run add:service -- support --description "Own support case resolution" --non-interactive
npm run add:agent -- classify-support-case --service support --service-version 1 --http none --description "Classify one support case" --non-interactive
npm run add:agent -- plan-support-resolution --service support --service-version 1 --http none --description "Plan one support resolution" --non-interactive
npm run add:workflow -- resolve-support-case --service support --service-version 1 --description "Run recoverable support resolution stages" --non-interactive
npm run add:command -- run-resolve-support-case --service support --service-version 1 --description "Run a durable multi-step support resolution" --non-interactive
```

This chapter supplies focused final tests and a smaller service surface. Remove the generated placeholders that are not part of the final project.

```bash title="Remove generated placeholders" replay="project"
rm src/service/support/generalSupportServiceInfo.ts
rm src/service/support/v1/supportServiceConfig.ts
rm src/service/support/v1/supportV1Service.test.ts
rm src/service/support/v1/harness/agent/classifySupportCase/classifySupportCaseAgent.test.ts
rm src/service/support/v1/harness/agent/planSupportResolution/planSupportResolutionAgent.test.ts
rm src/service/support/v1/harness/workflow/resolveSupportCase/resolveSupportCaseWorkflow.test.ts
rm src/service/support/v1/command/runResolveSupportCase/runResolveSupportCaseCommandBuilder.test.ts
```
