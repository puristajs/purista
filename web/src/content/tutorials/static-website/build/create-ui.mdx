## Install the browser packages

```bash title="Install React and the shadcn component helpers" replay="project"
npm install --save-exact react@19.2.8 react-dom@19.2.8 class-variance-authority@0.7.1 clsx@2.1.1 tailwind-merge@3.6.0 lucide-react@1.32.0
```

```bash title="Install the Vite, Tailwind, and browser test tools" replay="project"
npm install --save-dev --save-exact vite@8.2.1 @vitejs/plugin-react@5.2.0 tailwindcss@4.3.3 @tailwindcss/vite@4.3.3 @types/react@19.2.18 @types/react-dom@19.2.4 @testing-library/react@16.3.2 @testing-library/jest-dom@6.9.1 jsdom@30.0.1
```

Add separate scripts for the server and browser builds. `npm test` builds the
browser first because the server tests will read the real production files.

```bash title="Add browser build and test scripts" replay="project"
npm pkg set 'scripts.build:server=tsc' 'scripts.build:ui=vite build --config ui/vite.config.ts' 'scripts.build=npm run build:server && npm run build:ui' 'scripts.test:server=tsc --noEmit && vitest run --config vitest.server.config.ts' 'scripts.test:ui=vitest run --config ui/vitest.config.ts' 'scripts.test=npm run build:ui && npm run test:server && npm run test:ui'
```

Keep server discovery inside the server source tree. The browser gets its own
configuration on the testing page.

```ts title="vitest.server.config.ts" write
import { defineConfig } from 'vitest/config'

export default defineConfig({
	test: {
		environment: 'node',
		include: ['src/**/*.test.ts'],
	},
})
```

## Configure Vite and shadcn

Vite uses `ui/` as its root and writes production files to `ui/dist/`.

```ts title="ui/vite.config.ts" write
import tailwindcss from '@tailwindcss/vite'
import react from '@vitejs/plugin-react'
import { fileURLToPath, URL } from 'node:url'
import { defineConfig } from 'vite'

export default defineConfig({
	root: fileURLToPath(new URL('.', import.meta.url)),
	plugins: [react(), tailwindcss()],
	resolve: {
		alias: { '@': fileURLToPath(new URL('./src', import.meta.url)) },
	},
	build: { outDir: 'dist', emptyOutDir: true },
})
```

## Add the shared component helpers

These small files follow the shadcn component pattern. `cn` combines Tailwind
classes, while the component files keep recurring markup consistent.

```ts title="ui/src/lib/utils.ts" write
import { clsx, type ClassValue } from 'clsx'
import { twMerge } from 'tailwind-merge'

export function cn(...inputs: ClassValue[]) {
	return twMerge(clsx(inputs))
}
```

```tsx title="ui/src/components/ui/card.tsx" write
import type * as React from 'react'
import { cn } from '@/lib/utils'

export function Card({ className, ...props }: React.ComponentProps<'section'>) {
	return <section className={cn('rounded-xl border bg-card text-card-foreground shadow-sm', className)} {...props} />
}

export function CardHeader({ className, ...props }: React.ComponentProps<'div'>) {
	return <div className={cn('grid gap-1.5 p-6', className)} {...props} />
}

export function CardTitle({ className, ...props }: React.ComponentProps<'h2'>) {
	return <h2 className={cn('font-semibold leading-none tracking-tight', className)} {...props} />
}

export function CardDescription({ className, ...props }: React.ComponentProps<'p'>) {
	return <p className={cn('text-sm text-muted-foreground', className)} {...props} />
}

export function CardContent({ className, ...props }: React.ComponentProps<'div'>) {
	return <div className={cn('p-6 pt-0', className)} {...props} />
}
```

```tsx title="ui/src/components/ui/button.tsx" write
import { cva, type VariantProps } from 'class-variance-authority'
import type * as React from 'react'
import { cn } from '@/lib/utils'

export const buttonVariants = cva(
	'inline-flex items-center justify-center gap-2 rounded-md text-sm font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring disabled:pointer-events-none disabled:opacity-50',
	{
		variants: {
			variant: {
				default: 'bg-primary text-primary-foreground hover:bg-primary/90',
				outline: 'border bg-background hover:bg-accent hover:text-accent-foreground',
			},
			size: { default: 'h-10 px-4 py-2', sm: 'h-9 rounded-md px-3' },
		},
		defaultVariants: { variant: 'default', size: 'default' },
	},
)

type ButtonProps = React.ComponentProps<'button'> & VariantProps<typeof buttonVariants>

export function Button({ className, variant, size, ...props }: ButtonProps) {
	return <button className={cn(buttonVariants({ variant, size }), className)} {...props} />
}
```

## Create the read-only shell

```tsx title="ui/src/main.tsx" write
import { StrictMode } from 'react'
import { createRoot } from 'react-dom/client'
```
