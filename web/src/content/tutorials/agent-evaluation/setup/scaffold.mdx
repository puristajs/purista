Create a fresh project and enter it.

```bash title="Create and enter the evaluation project" replay="parent"
npm create purista@2.2.0 agent-evaluation -- --runtime node --event-bridge default --non-interactive --defaults
cd agent-evaluation
```

Use the published Framework, Harness, and provider packages. The default
runner added later remains credential-free.

```json title="package.json" write
{
	"name": "@purista/tutorial-agent-evaluation",
	"version": "1.0.0",
	"private": true,
	"type": "module",
	"engines": {
		"node": ">=24.15.0"
	},
	"scripts": {
		"add:service": "purista add service",
		"add:command": "purista add command",
		"add:agent": "purista add agent",
		"build": "tsc --noEmit",
		"demo": "node --import tsx src/demo.ts",
		"evaluate": "node --import tsx src/runEvaluation.ts",
		"evaluate:live": "node --env-file-if-exists=.env --import tsx src/runLiveEvaluation.ts",
		"lint": "biome check .",
		"start": "node --env-file-if-exists=.env --import tsx src/index.ts",
		"test": "vitest run"
	},
	"dependencies": {
		"@purista/core": "4.0.0",
		"@purista/harness": "3.0.0",
		"@purista/harness-openai": "3.0.0",
		"zod": "4.4.3"
	},
	"devDependencies": {
		"@biomejs/biome": "2.5.8",
		"@purista/cli": "4.0.0",
		"@types/node": "26.2.0",
		"@types/sinon": "22.0.0",
		"sinon": "22.1.0",
		"tsx": "4.23.12",
		"typescript": "7.0.2",
		"vitest": "4.1.10"
	}
}
```

```bash title="Install the evaluation dependencies" replay="project"
npm install
```

Generate the Support service, native Harness agent, and PURISTA command. The
evaluation must run this same agent definition rather than a simplified copy.

```bash title="Generate the evaluated boundaries" replay="project"
npm run add:service -- support --description "Own support classification" --non-interactive
npm run add:agent -- classify-support-message --service support --service-version 1 --description "Classify one support message" --non-interactive
npm run add:command -- classify-support-message --service support --service-version 1 --description "Classify one support message" --non-interactive
```

```bash title="Check the generated project"
npm run build
```
